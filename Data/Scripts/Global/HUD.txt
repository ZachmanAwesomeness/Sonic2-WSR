private alias object.value0 : object.savedYellow
private alias object.value1 : object.flashingTimer

private alias object.type   : player.type

// Game Mode Aliases
private alias 2 : MODE_TIMEATTACK

private alias 22 : GLOBAL_PLAYERSCORE

// ========================
// Function Declarations
// ========================

reserve function HUD_ApplyRed
reserve function HUD_ApplyYellow
reserve function HUD_Draw

// ========================
// Function Definitions
// ========================

private function HUD_ApplyYellow
	SetPaletteEntry(0, 15, object.savedYellow)
end function

private function HUD_ApplyRed
	GetPaletteEntry(0, 12, temp1)
	SetPaletteEntry(0, 15, temp1)
end function

private function HUD_Draw
	temp0 = 9
	DrawNumbers(5, 62, temp0, player.score, 6, 8, true)
	temp0 += 15

	// Draw Clock Icon
	DrawSpriteScreenXY(0, 28, temp0)
	temp0 += 3

	if stage.minutes == 9
		if object.flashingTimer > 8
			CallFunction(HUD_ApplyRed)
		else
			CallFunction(HUD_ApplyYellow)
		end if
	end if

	// Draw milliseconds alongside of the other values
	DrawSpriteScreenXY(16, 49, temp0)
	DrawNumbers(5, 91, temp0, stage.milliSeconds, 2, 8, true)
	DrawNumbers(5, 62, temp0, stage.seconds, 2, 8, true)
	DrawNumbers(5, 38, temp0, stage.minutes, 1, 8, true)
	temp0 += 17
	CallFunction(HUD_ApplyYellow)

	// Draw Ring Icon
	DrawSpriteScreenXY(1, 28, temp0)
	temp0 += 3

	if player[0].rings == 0
		if object.flashingTimer > 8
			CallFunction(HUD_ApplyRed)
		end if
	end if
	DrawNumbers(5, 54, temp0, player[0].rings, 3, 8, true)
	CallFunction(HUD_ApplyYellow)

	temp0 += 18

	// Character Icon
	temp1 = stage.playerListPos
	temp1 += 17
	DrawSpriteScreenXY(temp1, 28, temp0)
	temp0++
	if player.lives > 99
		DrawNumbers(5, 54, temp0, player.lives, 3, 8, true)
	else
		DrawNumbers(5, 46, temp0, player.lives, 2, 8, true)
	end if
end function

// ========================
// Events
// ========================

event ObjectDraw
	// Update flashing red texts
	if player[0].rings == 0
		object.flashingTimer++

		if object.flashingTimer == 16
			object.flashingTimer = 0
		end if
	else
		if stage.minutes == 9
			object.flashingTimer++

			if object.flashingTimer == 16
				object.flashingTimer = 0
			end if
		end if
	end if

	CallFunction(HUD_Draw)
end event


event ObjectStartup
	LoadSpriteSheet("Global/HUD.gif")

	SpriteFrame(-7, 0, 15, 18, 1, 1) 	// 0 - Clock
	SpriteFrame(-9, 0, 18, 19, 17, 1) 	// 1 - Ring Frame 1
	SpriteFrame(-7, 0, 14, 19, 1, 20) 	// 2 - Ring Frame 2
	SpriteFrame(-4, 0, 8, 19, 16, 21) 	// 3 - Ring Frame 3
	SpriteFrame(-7, 0, 14, 19, 25, 21) 	// 4 - Ring Frame 4

	temp0 = 20
	while temp0 < 47
		temp1 = 40
		while temp1 < 89
			SpriteFrame(0, 0, 9, 13, temp1, temp0) // 5-14 - Numbers
			temp1 += 10
		loop
		temp0 += 14
	loop

	SpriteFrame(0, 0, 5, 11, 93, 2) 	// 15 - :
	SpriteFrame(-1, 0, 34, 6, 1, 41) 	// 16 - ' "

	// Character icons
	SpriteFrame(-8, 0, 17, 16, 36, 1) 	// 17 - Sonic
	SpriteFrame(-9, 0, 18, 17, 54, 1) 	// 18 - Tails
	SpriteFrame(-10, 0, 19, 18, 73, 1) 	// 19 - Knuckles

	foreach(TypeName[HUD], arrayPos0, ALL_ENTITIES)
		GetPaletteEntry(0, 15, temp0)
		object[arrayPos0].savedYellow = temp0
	next
end event
